cmake_minimum_required(VERSION 3.18)
project(multispectral_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Threads REQUIRED)
find_path(CV_INCLUDE opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
find_path(JSON_INCLUDE nlohmann/json.hpp REQUIRED)
find_path(HTTP_INCLUDE httplib.h REQUIRED)
find_library(HTTP_LIBRARY NAMES cpp-httplib REQUIRED)
foreach(module core imgproc imgcodecs video videoio calib3d features2d flann)
  find_library(CV_${module} NAMES opencv_${module} REQUIRED)
  list(APPEND CV_LIBS ${CV_${module}})
endforeach()
option(IMAGING_ENABLE_VULKAN "Build Vulkan compute backend when dependencies exist" ON)
set(GPU_SOURCE vulkan_stub.cpp)
if(IMAGING_ENABLE_VULKAN)
  find_path(VULKAN_INCLUDE vulkan/vulkan.h)
  find_library(VULKAN_LIBRARY NAMES vulkan)
  find_program(GLSLANG_VALIDATOR glslangValidator)
  if(VULKAN_INCLUDE AND VULKAN_LIBRARY AND GLSLANG_VALIDATOR)
    set(GPU_SOURCE vulkan_remap.cpp)
    set(HAVE_VULKAN TRUE)
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/local_remap_spirv.hpp
      COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.0
              ${CMAKE_CURRENT_SOURCE_DIR}/shaders/local_remap.comp
              -o ${CMAKE_CURRENT_BINARY_DIR}/local_remap.spv
      COMMAND ${CMAKE_COMMAND} -DINPUT=${CMAKE_CURRENT_BINARY_DIR}/local_remap.spv
              -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/local_remap_spirv.hpp
              -P ${CMAKE_CURRENT_SOURCE_DIR}/embed_spirv.cmake
      DEPENDS shaders/local_remap.comp embed_spirv.cmake VERBATIM)
    list(APPEND GPU_SOURCE ${CMAKE_CURRENT_BINARY_DIR}/local_remap_spirv.hpp)
    message(STATUS "Vulkan compute backend enabled")
  else()
    message(STATUS "Vulkan dependencies unavailable: building CPU backend with explicit fallback status")
  endif()
endif()
add_library(imaging_core core.cpp ${GPU_SOURCE})
if(HAVE_VULKAN)
  target_include_directories(imaging_core PRIVATE ${VULKAN_INCLUDE} ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(imaging_core PUBLIC ${VULKAN_LIBRARY})
  add_executable(vulkan_test vulkan_test.cpp)
  target_link_libraries(vulkan_test PRIVATE imaging_core)
endif()
target_include_directories(imaging_core PUBLIC ${CV_INCLUDE} ${JSON_INCLUDE})
target_link_libraries(imaging_core PUBLIC ${CV_LIBS} Threads::Threads)
target_compile_options(imaging_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(imaging_cpp main.cpp runtime.cpp)
target_include_directories(imaging_cpp PRIVATE ${HTTP_INCLUDE})
target_link_libraries(imaging_cpp PRIVATE imaging_core ${HTTP_LIBRARY})
target_compile_definitions(imaging_cpp PRIVATE DEFAULT_WEB="${CMAKE_CURRENT_SOURCE_DIR}/web/index.html")
target_compile_options(imaging_cpp PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(core_test core_test.cpp)
target_link_libraries(core_test PRIVATE imaging_core)
add_test(NAME imaging_core_test COMMAND core_test)

if(HAVE_VULKAN)
  add_test(NAME vulkan_parity COMMAND vulkan_test)
  set_tests_properties(vulkan_parity PROPERTIES SKIP_RETURN_CODE 77)
endif()

add_executable(cpu_parity_test cpu_parity_test.cpp)
target_link_libraries(cpu_parity_test PRIVATE imaging_core)
add_test(NAME cpu_parity COMMAND cpu_parity_test)
